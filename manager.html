// 알림 객체를 순서대로 보관할 배열 (최대 3개 유지) let notificationQueue = []; function showDesktopNotification(carData) { if ("Notification" in window && Notification.permission === "granted") { // 1. 화면에 떠있는 알림이 3개 이상이면 가장 오래된 알림(첫 번째)을 닫고 배열에서 제거 if (notificationQueue.length >= 3) { const oldestNotification = notificationQueue.shift(); try { oldestNotification.close(); } catch(e) {} } const title = `🚨 [신규 차량 입문] ${carData.car_number || '차량'}`; const options = { body: `운전자: ${carData.driver_name || '미기재'} | 연락처: ${carData.phone_number || '미기재'}`, icon: 'https://cdn-icons-png.flaticon.com/512/2555/2555013.png', // 고유 tag를 사용하지 않아야 각 알림이 카드로 독립 생성되어 stacked(최대 3개)로 쌓임 tag: 'shc-dock-' + Date.now(), requireInteraction: true // 사용자가 닫거나 스크립트가 닫을 때까지 화면에 유지 }; // 2. 새 알림 생성 후 큐(배열)에 저장 const newNoti = new Notification(title, options); newNoti.onclick = () => { window.focus(); }; notificationQueue.push(newNoti); } }
